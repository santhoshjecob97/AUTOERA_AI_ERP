```python
class FeatureStore:
    def __init__(self):
        self.features = {}
        self.metadata = {}
    
    def register_feature(self, name, feature_function, description):
        """Register a new feature extraction function"""
        self.features[name] = feature_function
        self.metadata[name] = {
            'description': description,
            'created_at': datetime.now(),
            'version': 1
        }
    
    def get_features(self, data, feature_names):
        """Extract specified features from data"""
        extracted_features = {}
        for name in feature_names:
            if name in self.features:
                extracted_features[name] = self.features[name](data)
        
        return extracted_features
    
    def update_feature(self, name, new_function):
        """Update feature extraction logic"""
        if name in self.features:
            self.features[name] = new_function
            self.metadata[name]['version'] += 1
            self.metadata[name]['updated_at'] = datetime.now()
```
